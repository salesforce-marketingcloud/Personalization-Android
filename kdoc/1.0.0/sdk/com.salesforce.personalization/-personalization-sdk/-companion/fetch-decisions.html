fetchDecisions

suspend fun fetchDecisions(personalizationPointNames: Array<String>, decisionsRequestContext: DecisionsRequestContext? = null, timeoutMs: Long = TimeoutManager.Defaults.OVERALL_TIMEOUT): Result<DecisionsResponse, PersonalizationException>

Fetch personalization decisions for the provided Personalization Points and optional context.

Recommended Kotlin suspend function that handles coroutine scope management internally. Respects your calling scope's lifecycle - if cancelled, this operation will be cancelled too. Safe to call from any thread; network work happens on background threads.

Usage:

// In Compose (recommended approach)
@Composable
fun PersonalizedContent() {
    var personalizations by remember { mutableStateOf<DecisionsResponse?>(null) }
    var isLoading by remember { mutableStateOf(false) }

    // Option 1: Using LaunchedEffect
    LaunchedEffect(Unit) {
        isLoading = true
        val result = PersonalizationSdk.fetchDecisions(arrayOf("homepage_banner"))
        when (result) {
            is Result.Success -> personalizations = result.output
            is Result.Failure -> handleError(result.throwable)
        }
        isLoading = false
    }

    // Option 2: Using rememberCoroutineScope with user interaction
    val scope = rememberCoroutineScope()
    Button(onClick = {
        scope.launch {
            val result = PersonalizationSdk.fetchDecisions(arrayOf("product_recommendations"))
            when (result) {
                is Result.Success -> personalizations = result.output
                is Result.Failure -> handleError(result.throwable)
            }
        }
    }) { Text("Load Personalizations") }
}

// In ViewModel
viewModelScope.launch {
    val result = PersonalizationSdk.fetchDecisions(arrayOf("recommendations"))
    // Handle result...
}
